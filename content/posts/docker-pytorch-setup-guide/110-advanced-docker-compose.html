Advanced: Docker Compose

For more complex setups, use Docker Compose. Create docker-compose.yml:

version: '3.8'

services:
  pytorch-training:
    build: .
    volumes:
      - ./data:/app/data
      - ./models:/app/models
      - ./logs:/app/logs
    environment:
      - PYTHONUNBUFFERED=1
    command: python train.py

Run with:

docker-compose up
